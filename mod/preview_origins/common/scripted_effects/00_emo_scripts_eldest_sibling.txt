# scope = country
terraform_planet_and_generate_late_pre_ftl_empire = {
	# prefer habitable planets and terraforming candidates
	if = {
		limit = {
			any_planet_within_border = {
				OR = {
					AND = {
						is_colony = no
						habitable_planet = yes
					}
					has_modifier = terraforming_candidate
					has_modifier = toxic_terraforming_candidate
				}
			}
		}
		random_planet_within_border = {
			limit = {
				OR = {
					AND = {
						is_colony = no
						habitable_planet = yes
					}
					has_modifier = terraforming_candidate
					has_modifier = toxic_terraforming_candidate
				}
			}
			generate_late_primitives_on_planet = yes
		}
	}
	else = {
		random_planet_within_border = {
			limit = {
				# limits cannot be scripted effects and need to be duplicated code
				is_colony = no
				planet_size < 26
				is_star = no
				is_asteroid = no
				NOT = { is_planet_class = pc_gas_giant }
				NOT = { has_orbital_station = yes }
			}
			if = {
				limit = { has_global_flag = planetarydiversity }
				change_pc = rl_habitable_normal
				reroll_deposits = yes
				reroll_planet_modifiers = yes
				reroll_pd_class_modifiers = yes
			}
			else = {
				change_pc = rl_habitable_normal
				reroll_deposits = yes
				reroll_planet_modifiers = yes
			}
			generate_late_primitives_on_planet = yes
		}
	}
}

# Merges and optimizes choose_random_planet_to_terraform_and_generate_same_species_primitives, terraform_planet_and_generate_same_species_primitives, and reroll_planet_and_generate_same_species_primitives into one trigger
# Similar optimizations can probably be made to the PD compat
# scope = capital
choose_random_planet_to_terraform_and_generate_same_species_primitives = {
	solar_system = {
		# prefer habitable planets and terraforming candidates
		if = {
			limit = {
				OR = {
					any_system_planet = {
						is_colony = no
						habitable_planet = yes
					}
					any_system_planet = {
						has_modifier = terraforming_candidate
					}
					any_system_planet = {
						has_modifier = toxic_terraforming_candidate
					}
				}
			}
			random_system_planet = {
				limit = {
					OR = {
						AND = {
							is_colony = no
							habitable_planet = yes
						}
						has_modifier = terraforming_candidate
						has_modifier = toxic_terraforming_candidate
					}
				}
				generate_same_species_late_primitives_on_planet = yes
			}
		}
		else = {
			random_system_planet = {
				limit = {
					is_colony = no
					planet_size < 26
					is_star = no
					is_asteroid = no
					NOT = { is_planet_class = pc_gas_giant }
					NOT = { has_orbital_station = yes }
				}
				root = {
					capital_scope = {
						switch = {
							trigger = is_planet_class
							pc_continental = { prevprev = { change_pc = pc_continental } }
							pc_ocean       = { prevprev = { change_pc = pc_ocean } }
							pc_tropical    = { prevprev = { change_pc = pc_tropical } }
							pc_alpine      = { prevprev = { change_pc = pc_alpine } }
							pc_arctic      = { prevprev = { change_pc = pc_arctic } }
							pc_desert      = { prevprev = { change_pc = pc_desert } }
							pc_savannah    = { prevprev = { change_pc = pc_savannah } }
							pc_tundra      = { prevprev = { change_pc = pc_tundra } }
							pc_arid        = { prevprev = { change_pc = pc_arid } }
						}
					}
				}
				# If I understand the code correctly, these checks and calls should be outside of the root/capital_scope/switch block for correct scope purposes
				if = {
					limit = { has_modifier = terraforming_candidate	}
					remove_modifier = terraforming_candidate
				}
				if = {
					limit = { has_modifier = toxic_terraforming_candidate }
					remove_modifier = toxic_terraforming_candidate
				}
				reroll_deposits = yes
				reroll_planet_modifiers = yes
				generate_same_species_late_primitives_on_planet = yes
			}
		}
	}
}

# scope = Non-Colonized Habitable Planet
# adapted from /Stellaris/common/scripted_effects/pre_ftl_scripted_effects.txt 
generate_same_species_late_primitives_on_planet = {
	optimize_memory
	# create new species based on the original one
	modify_species = {
		species = space_owner.species
	}
	# randomize whether to keep or gain subterranean origin and trait
	random_list = {
		# use 1:4 odds in either case
		10 = {
			modifier = {
				factor = 16
				space_owner.species = {
					has_trait = trait_cave_dweller
				}
			}
			set_planet_flag = pre_ftl_subterranean
			if = {
				limit = {
					space_owner.species = {
						NOT = {
							has_trait = trait_cave_dweller
						}
					}
				}
				modify_species = {
					species = last_created_species
					add_traits_at_start_of_list = yes
					add_trait = trait_cave_dweller
				}
			}
		}
		40 = {
			set_default_origin = yes
			if = {
				limit = {
					space_owner.species = {
						has_trait = trait_cave_dweller
					}
				}
				modify_species = {
					species = last_created_species
					remove_trait = trait_cave_dweller
				}
			}
		}
	}
	# randomize whether to keep or gain hive mind trait
	random_list = {
		# use 1:9 odds in either case
		10 = {
			modifier = {
				factor = 81
				space_owner.species = {
					has_trait = trait_hive_mind
				}
			}
			# create hive mind pre ftl empire
			if = {
				limit = {
					space_owner.species = {
						NOT = {
							has_trait = trait_hive_mind
						}
					}
				}
				modify_species = {
					species = last_created_species
					add_traits_at_start_of_list = yes
					add_trait = trait_hive_mind
				}
			}
			create_late_pre_ftl_hive_empire = yes
			setup_pre_ftl_hive_planet = yes
			# pre_ftl_clean_up_effect = yes # missing in vanilla?
		}
		90 = {
			# create default pre ftl empire
			if = {
				limit = {
					space_owner.species = {
						has_trait = trait_hive_mind
					}
				}
				modify_species = {
					species = last_created_species
					remove_trait = trait_hive_mind
				}
			}
			create_late_pre_ftl_empire = yes
			setup_pre_ftl_planet = yes
			pre_ftl_clean_up_effect = yes
		}
	}
}
